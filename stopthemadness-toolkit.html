StopTheMadness Safari Extension Toolkit

Open this page in Safari, then copy the snippets into StopTheMadness Pro. Use the CSS blocks in Custom CSS, the JavaScript blocks in JavaScript snippets for manual actions, or in Custom JavaScript for automatic page-load behavior.

How to use

  1. In StopTheMadness, open the website options for a domain or leave defaults selected.
  2. Paste one CSS block into Custom CSS.
  3. Add JavaScript snippets one at a time in JavaScript snippets; enable “Run snippet only in the top frame” for snippets that change the whole page.
  4. Keep aggressive snippets domain-specific so they do not break login, checkout, banking, or editor pages.

Custom CSS blocks

Reader-ish cleanup

Hides sticky bars, cookie prompts, newsletter overlays, and common ad containers.

/* General calmer web: use per-site if a page breaks. */
:where([class*="cookie" i],[id*="cookie" i],[class*="newsletter" i],[id*="newsletter" i],[class*="subscribe" i],[id*="subscribe" i],[class*="modal" i],[id*="modal" i],[class*="paywall" i],[id*="paywall" i],[aria-modal="true"]) {
  display: none !important;
}
:where([class*="sticky" i],[id*="sticky" i],[class*="floating" i],[id*="floating" i],.ad,.ads,[class*=" ad-" i],[class^="ad-" i],[id^="ad-" i],[class*="sponsor" i]) {
  display: none !important;
}
html, body { overflow: auto !important; }

JavaScript snippets

Remove overlays and unlock scroll

(() => {
  const selectors = ['[aria-modal="true"]','[role="dialog"]','[class*="modal" i]','[id*="modal" i]','[class*="overlay" i]','[id*="overlay" i]','[class*="cookie" i]','[id*="cookie" i]','[class*="newsletter" i]','[id*="newsletter" i]'];
  document.querySelectorAll(selectors.join(',')).forEach(el => el.remove());
  document.documentElement.style.overflow = 'auto';
  document.body.style.overflow = 'auto';
})();

Show all hidden images

(() => {
  document.querySelectorAll('img').forEach(img => {
    for (const attr of ['data-src','data-original','data-lazy-src','data-url','data-image']) {
      if (!img.src && img.getAttribute(attr)) img.src = img.getAttribute(attr);
    }
    img.loading = 'eager';
    img.decoding = 'auto';
    img.style.visibility = 'visible';
    img.style.opacity = '1';
  });
})();

Copy clean page link

(() => {
  const url = new URL(location.href);
  [...url.searchParams.keys()].forEach(k => {
    if (/^(utm_|fbclid$|gclid$|msclkid$|mc_|igshid$|ref$|spm$)/i.test(k)) url.searchParams.delete(k);
  });
  navigator.clipboard.writeText(url.toString());
})();

Toggle editable page text

(() => {
  document.designMode = document.designMode === 'on' ? 'off' : 'on';
  alert(`Page editing is now ${document.designMode}. Reload to discard edits.`);
})();

Expand details and collapsed content

(() => {
  document.querySelectorAll('details').forEach(d => d.open = true);
  document.querySelectorAll('[aria-expanded="false"]').forEach(el => el.setAttribute('aria-expanded','true'));
  document.querySelectorAll('[hidden], [style*="display: none"]').forEach(el => {
    el.hidden = false;
    el.style.display = 'block';
  });
})();

Make videos usable

(() => {
  document.querySelectorAll('video,audio').forEach(media => {
    media.controls = true;
    media.muted = false;
    media.playbackRate = 1;
    media.removeAttribute('disablepictureinpicture');
  });
})();
